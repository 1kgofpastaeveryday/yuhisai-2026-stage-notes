cmake_minimum_required(VERSION 3.21)
project(qlcplus_name_sync LANGUAGES C CXX)
set(QLC_SOURCE "" CACHE PATH "Pinned QLC+ 5.3.0 source with stability.patch applied")
set(QLC_RUNTIME "/Applications/QLC+ 5.app/Contents" CACHE PATH "Unmodified QLC+ 5.3.0 runtime")
set(CMAKE_INCLUDE_CURRENT_DIR ON)
set(INSTALL_ROOT "/")
set(qmlui ON)
find_package(Qt6 6.10.3 EXACT REQUIRED COMPONENTS Core Gui Multimedia MultimediaWidgets Network PrintSupport Qml Quick Svg Test Widgets LinguistTools 3DCore 3DInput 3DLogic 3DQuick 3DQuickExtras 3DRender WebSockets)
set(QT_VERSION_MAJOR 6)
include("${QLC_SOURCE}/variables.cmake")
configure_file("${QLC_SOURCE}/engine/src/qlcconfig.h.noroot.in" "${CMAKE_CURRENT_BINARY_DIR}/qlcconfig.h")
add_library(qlcplusengine SHARED IMPORTED GLOBAL)
set_target_properties(qlcplusengine PROPERTIES
    IMPORTED_LOCATION "${QLC_RUNTIME}/Frameworks/libqlcplusengine.dylib"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR};${QLC_SOURCE}/plugins/interfaces;${QLC_SOURCE}/hotplugmonitor/src;${QLC_SOURCE}/engine/audio/src")
add_subdirectory("${QLC_SOURCE}/webaccess/src" webaccess)
add_subdirectory("${QLC_SOURCE}/qmlui" qmlui)

# Test the actual frontend backend, using the installed engine and I/O plugins.
# Reuse the upstream file list, excluding the application's main entry point.
get_target_property(frontend_sources qlcplus5 SOURCES)
set(test_sources "${CMAKE_CURRENT_SOURCE_DIR}/stability-test.cpp")
foreach(src IN LISTS frontend_sources)
    if(src MATCHES "(^|/)main\\.cpp$" OR src MATCHES "\\.(qm|rc)$")
        continue()
    endif()
    if(NOT IS_ABSOLUTE "${src}")
        set(src "${QLC_SOURCE}/qmlui/${src}")
    endif()
    list(APPEND test_sources "${src}")
endforeach()
add_executable(stability-test ${test_sources})
get_target_property(frontend_includes qlcplus5 INCLUDE_DIRECTORIES)
set(test_includes "${QLC_SOURCE}/qmlui")
foreach(inc IN LISTS frontend_includes)
    if(NOT IS_ABSOLUTE "${inc}")
        set(inc "${QLC_SOURCE}/qmlui/${inc}")
    endif()
    list(APPEND test_includes "${inc}")
endforeach()
target_include_directories(stability-test PRIVATE ${test_includes})
get_target_property(frontend_defs qlcplus5 COMPILE_DEFINITIONS)
target_compile_definitions(stability-test PRIVATE ${frontend_defs})
get_target_property(frontend_libs qlcplus5 LINK_LIBRARIES)
target_link_libraries(stability-test PRIVATE ${frontend_libs} Qt6::Test)

add_executable(qt-gc-test qt-gc-test.cpp)
target_link_libraries(qt-gc-test PRIVATE Qt6::Core Qt6::Qml)

add_executable(qt-canvas-test qt-canvas-test.cpp)
target_link_libraries(qt-canvas-test PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
