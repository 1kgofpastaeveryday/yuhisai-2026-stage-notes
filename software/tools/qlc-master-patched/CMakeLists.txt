get_target_property(frontend_sources qlcplus5 SOURCES)
get_target_property(frontend_includes qlcplus5 INCLUDE_DIRECTORIES)
get_target_property(frontend_definitions qlcplus5 COMPILE_DEFINITIONS)
get_target_property(frontend_libraries qlcplus5 LINK_LIBRARIES)
set(shared_sources)
set(application_sources)
foreach(source IN LISTS frontend_sources)
    if(source MATCHES "(^|/)main\\.cpp$" OR source MATCHES "\\.(qm|rc)$" OR source MATCHES "(^|/)qrc_[^/]*\\.cpp$")
        if(NOT IS_ABSOLUTE "${source}")
            set(source "${PROJECT_SOURCE_DIR}/qmlui/${source}")
        endif()
        list(APPEND application_sources "${source}")
        continue()
    endif()
    if(NOT IS_ABSOLUTE "${source}")
        set(source "${PROJECT_SOURCE_DIR}/qmlui/${source}")
    endif()
    list(APPEND shared_sources "${source}")
endforeach()
add_library(candidate_frontend OBJECT ${shared_sources})
target_include_directories(candidate_frontend PUBLIC "${PROJECT_SOURCE_DIR}/qmlui" ${frontend_includes})
target_compile_definitions(candidate_frontend PUBLIC ${frontend_definitions})
target_link_libraries(candidate_frontend PUBLIC ${frontend_libraries})
# Build the application and tests from the same frontend object files.
set_property(TARGET qlcplus5 PROPERTY SOURCES ${application_sources})
target_link_libraries(qlcplus5 PRIVATE candidate_frontend)
qt_add_resources(test_resources "${PROJECT_SOURCE_DIR}/qmlui/qmlui.qrc"
    "${PROJECT_SOURCE_DIR}/resources/icons/svg/svgicons.qrc"
    "${PROJECT_SOURCE_DIR}/resources/fonts/fonts.qrc")
add_library(candidate_test_resources OBJECT ${test_resources})
target_link_libraries(candidate_test_resources PRIVATE Qt${QT_MAJOR_VERSION}::Core)
foreach(pair "sceneeditor_test|sceneeditor_test.cpp" "function_identity_test|function-identity-test.cpp" "local_features_test|name-sync-test.cpp" "save_regression_test|save-regression.cpp")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 target)
    list(GET parts 1 source)
    add_executable(${target} ${source} $<TARGET_OBJECTS:candidate_test_resources>)
    target_link_libraries(${target} PRIVATE candidate_frontend Qt${QT_MAJOR_VERSION}::Test)
endforeach()
# Upstream's Windows build hardcodes its CI SDK path. Point only this local
# build at the staged SDK and import library for the installed FTDI DLL.
get_target_property(dmx_includes dmxusb INCLUDE_DIRECTORIES)
get_target_property(dmx_libraries dmxusb LINK_LIBRARIES)
string(REPLACE "<PATH>" "<PATH>" dmx_includes "${dmx_includes}")
string(REPLACE "<PATH>" "<PATH>" dmx_libraries "${dmx_libraries}")
set_property(TARGET dmxusb PROPERTY INCLUDE_DIRECTORIES ${dmx_includes})
set_property(TARGET dmxusb PROPERTY LINK_LIBRARIES ${dmx_libraries})

target_link_libraries(integration_host PRIVATE candidate_frontend)
